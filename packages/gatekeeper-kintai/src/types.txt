// Kintai (勤怠) is the company attendance ledger: clock punches, per-project hour allocations, and
// overtime requests routed to approvers through the organisation chart.
//
// The session is ambient and already scoped to ONE employee: the caller. Identity arrives out of
// band with the capability itself, so NO method takes an employee identifier, and there is no way
// to ask about somebody else. `listPendingApprovals()` is the one place another employee's data
// appears, and only because the caller is an approver for it in the org chart.
//
// Every timestamp is taken from the server clock. Work dates are JST (Asia/Tokyo, no DST) calendar
// dates in `YYYY-MM-DD` form. Which work date a punch is attributed to is a property of the
// EMPLOYEE, set by HR and not readable or settable from here:
//
// - by default, and for everyone who does not work nights, a punch belongs to the JST date it
//   happened on;
// - for an employee HR has put on shift-start dating, a punch belongs to the date of the shift
//   that is currently open, so a 22:00 -> 06:00 shift lands entirely on the date it started.
//
// Shift-start dating stops after SIXTEEN HOURS. A shift that has been open that long no longer
// attracts punches: the next punch is filed against its own calendar date and starts a fresh
// shift, and the abandoned day keeps its `unpaired_in` forever. That is the usual explanation when
// a night worker's `punch()` comes back with today's date instead of the shift's, and when a past
// day shows `unpaired_in` on someone who does not normally get one — somebody forgot to clock out.
// It cannot be repaired from here; an administrator has to correct the record.
//
// So never assume `punch()` returns today's date. Read `workDate` off the receipt and use THAT —
// it is the day the punch was actually filed against, and it is the argument `getDay()` wants.
//
// A new account is NOT usable until HR links it to an employee record. Until then `whoAmI()`
// returns `{ linked: false, employeeId: null }` and every other method throws
// `KINTAI_ACCOUNT_NOT_LINKED`. Call `whoAmI()` first and tell the user to contact HR rather than
// retrying.
//
// Errors carry a machine-readable code as the first token of the message, e.g.
// `KINTAI_PERIOD_LOCKED: 2026-07 is closed`. The codes are:
//
// - `KINTAI_ACCOUNT_NOT_LINKED` — this account has no employee record yet.
// - `KINTAI_INVALID_INPUT` — a malformed date, a negative/fractional minute count, or input over
//   a size limit (at most 200 allocation entries per day; project code 64 characters, note 500,
//   overtime reason 2000, approval comment 2000).
// - `KINTAI_PERIOD_LOCKED` — the month is closed for payroll; the record can no longer be edited.
// - `KINTAI_NOT_AUTHORIZED` — the caller is not an approver for this submission at its current
//   step. Also returned for a submission belonging to someone else, so it reveals nothing.
// - `KINTAI_SELF_APPROVAL` — nobody may act on their own submission.
// - `KINTAI_EXEMPT_EMPLOYEE` — 管理監督者 (a manager exempt from overtime premiums) cannot file
//   overtime.
// - `KINTAI_NO_ROUTE` / `KINTAI_NO_APPROVER` — no approval route matches, or nobody can approve.
// - `KINTAI_NOT_FOUND` — no such submission.
// - `KINTAI_INVALID_TRANSITION` — the submission is not in a state that allows this.
// - `KINTAI_WORK_DATE_RACED` — two punches for this employee landed at once and the day this one
//   belongs to changed underneath it. Nothing was recorded; see `punch()`.

/** A clock event. `break_start`/`break_end` bracket unpaid time inside a shift. */
export type PunchKind = "in" | "out" | "break_start" | "break_end";

/** How a punch's coordinates were obtained. */
export type LocationSource = "gps" | "denied" | "unavailable" | "manual";

/** Optional geolocation attached to a punch, used to match a registered work site. */
export type PunchLocation = {
  source: LocationSource;
  latitude?: number;
  longitude?: number;
  accuracyM?: number;
};

/**
 * One recorded punch, as stored. `occurred_at` and `recorded_at` differ only for administrative
 * corrections; a punch made through this session has them equal. A superseded punch is never
 * deleted — `supersedes_id` chains a correction back to what it replaced.
 *
 * The location fields (`latitude`, `longitude`, `accuracy_m`, `location_source`,
 * `matched_site_id`) are personal data kept on a SHORTER retention clock than the punch itself:
 * they can be purged while the attendance record stays. Read them as null for any punch made
 * without location, and do not rely on them still being there for an old punch.
 */
export type PunchRow = {
  id: number;
  employee_id: number;
  work_date: string;
  kind: PunchKind;
  occurred_at: number;
  recorded_at: number;
  source: "gadget" | "admin" | "import";
  latitude: number | null;
  longitude: number | null;
  accuracy_m: number | null;
  location_source: LocationSource | null;
  matched_site_id: number | null;
  supersedes_id: number | null;
  amended_by: number | null;
  amend_reason: string | null;
};

/** One line of a day's project allocation. `minutes` must be a whole, non-negative number. */
export type AllocationEntry = {
  projectCode: string;
  minutes: number;
  note?: string;
};

/** A stored allocation line. Editing a day supersedes the previous version rather than erasing it. */
export type AllocationRow = {
  id: number;
  employee_id: number;
  work_date: string;
  project_code: string;
  minutes: number;
  note: string | null;
  version: number;
  superseded_by: number | null;
};

/**
 * How the day's allocated minutes compare with the minutes actually worked from the punch record.
 * A discrepancy is information, never a rejection: `setAllocations()` succeeds either way.
 */
export type Reconciliation = {
  allocatedMinutes: number;
  workedMinutes: number;
  /** `allocatedMinutes - workedMinutes`. Negative means the day is under-allocated. */
  discrepancyMinutes: number;
};

/**
 * Everything known about one work date for the caller.
 *
 * `anomalies` explains a day whose `workedMinutes` looks wrong, and is why it travels with the
 * reconciliation: an unpaired clock-in contributes ZERO minutes rather than an open-ended run, so
 * without this the day would just look short with no reason attached. Values:
 *
 * - `unpaired_in` — clocked in and never out (the usual cause of a short day).
 * - `unpaired_break` — a break was started and never ended.
 * - `orphan_out` — clocked out with no matching clock-in.
 * - `duplicate_in` — clocked in twice without clocking out.
 * - `negative_gross` — punches are out of order; the record needs an administrative correction.
 * - `long_span` — clock-in to clock-out totals fourteen hours or more on this day. The punches are
 *   well-formed and the minutes are credited in full; this says the DAY is unusual. Almost always
 *   a clock-out that was hours late, occasionally a genuine long shift. Report the total AND this
 *   flag, and say it needs checking; never present it as an ordinary day. Applies to every
 *   employee on either work-date policy — the flag describes the punches, not the setting.
 *
 * NIGHT SHIFTS depend on the employee's work-date policy, which HR sets and this session cannot
 * see. For an employee on shift-start dating a shift crossing JST midnight is ONE work date with
 * the whole span credited, and normally no anomaly — but check `anomalies` rather than assuming
 * it is empty. A forgotten clock-out lands on that one day too: `long_span` for a very long
 * credited day, `duplicate_in` when a second clock-in arrived before the first was closed, and
 * `unpaired_in` once the sixteen-hour cutoff has passed and the shift was abandoned.
 *
 * For an employee on the default calendar dating — which is everyone who does not work nights —
 * such a shift is still split across two work dates, producing `unpaired_in` on the first and
 * `orphan_out` on the second: the first credits ZERO for the hours before midnight and the second
 * nothing for the hours after it, so the day reads as no work at all. You can recognise the case
 * from the anomalies themselves. When you see that pair on consecutive days, report the anomalies
 * and the raw punches rather than a minutes total, and say the total is unavailable rather than
 * zero — then tell the user their work-date policy may be wrong and that HR sets it.
 *
 * `locked` means the month is closed for payroll: reads still work, writes throw
 * `KINTAI_PERIOD_LOCKED`.
 */
export type KintaiDay = {
  punches: PunchRow[];
  allocations: AllocationRow[];
  reconciliation: Reconciliation;
  anomalies: string[];
  locked: boolean;
};

/** Lifecycle of an overtime request. `return` puts it back to `draft` for the employee to amend. */
export type SubmissionState = "draft" | "pending" | "approved" | "rejected" | "withdrawn";

/** What an approver can do to a submission at its current step. */
export type ApprovalAction = "approve" | "reject" | "return";

/**
 * An overtime request. `route_snapshot` is the approval route frozen as JSON at submission time, so
 * a later reorganisation cannot retroactively change who was required to approve; `current_step`
 * indexes into it.
 */
export type SubmissionRow = {
  id: number;
  employee_id: number;
  kind: "overtime";
  requested_for: string;
  state: SubmissionState;
  submitted_at: number | null;
  current_step: number;
  minutes: number;
  reason: string;
  calculation_inputs: string | null;
  route_snapshot: string;
  created_by: number | null;
};

/** Who the capability belongs to. `linked` is false until HR maps the account to an employee. */
export type KintaiIdentity = {
  linked: boolean;
  employeeId: number | null;
};

/**
 * Receipt for a recorded punch, including the JST work date it was attributed to.
 *
 * `workDate` is the server's answer and the only reliable one: for an employee on shift-start
 * dating it can be yesterday. Pass it straight to `getDay()`.
 */
export type PunchReceipt = {
  punchId: number;
  employeeId: number;
  workDate: string;
};

/**
 * The caller's own attendance record, plus the approval queue they are responsible for.
 *
 * This capability speaks for exactly one employee and cannot be pointed at another. Reads are
 * authorized as observations before any data is returned.
 */
export interface KintaiSession {
  /**
   * Which employee this capability speaks for.
   *
   * The only method that does not throw on an unlinked account — a new hire before HR sets them up
   * is an ordinary state the UI has to be able to explain. Call this first.
   */
  whoAmI(): Promise<KintaiIdentity>;

  /**
   * Record a clock event at the current server time.
   *
   * The work date it lands on is decided by the server from the employee's own work-date policy —
   * usually today's JST date, but the open shift's date for a night worker. It is on the receipt;
   * do not compute it yourself. There is no way to choose it, and no way to punch at another time.
   *
   * A repeat of the same kind within one minute is treated as a double-tap and returns the original
   * punch rather than creating a second one. Pass `location` only if the user has actually granted
   * geolocation; `{ source: "denied" }` records the refusal honestly.
   *
   * Throws `KINTAI_PERIOD_LOCKED` if the month the punch would land in is already closed — which,
   * for a night worker clocking out after midnight on the first, may not be this month.
   *
   * Throws `KINTAI_WORK_DATE_RACED` only if two punches for the same employee are recorded at the
   * same instant and the server could not agree twice running on which day this one belongs to.
   * It already retried once. Nothing was written, so it is safe to call again — but tell the user
   * rather than retrying in a loop, because a second failure means something else is punching for
   * this employee concurrently.
   */
  punch(kind: PunchKind, location?: PunchLocation): Promise<PunchReceipt>;

  /** Everything recorded for one JST work date: punches, allocations, reconciliation, anomalies. */
  getDay(workDate: string): Promise<KintaiDay>;

  /**
   * Replace the whole allocation set for one work date, superseding whatever was there.
   *
   * This is a full replacement, not a merge: entries you omit are dropped. Read `getDay()` first
   * and send back the complete list. Allocations need not sum to the minutes worked; the returned
   * reconciliation reports the difference rather than refusing.
   */
  setAllocations(workDate: string, entries: AllocationEntry[]): Promise<Reconciliation>;

  /**
   * File an overtime request for a past or current work date and return its submission id.
   *
   * The approval route is chosen from the employee's department and employment type and frozen onto
   * the submission. Filing is deliberately still allowed for a closed month — a submission is a
   * request for approval, not an edit to the ledger, so an employee who missed the cutoff can still
   * raise it.
   */
  submitOvertime(requestedFor: string, minutes: number, reason: string): Promise<number>;

  /** Take back one of the caller's own pending submissions. */
  withdrawSubmission(submissionId: number): Promise<void>;

  /** Send a returned (`draft`) submission back into the queue. The only way out of `draft`. */
  resubmit(submissionId: number): Promise<void>;

  /** The caller's own submissions, newest first. */
  listMySubmissions(): Promise<SubmissionRow[]>;

  /**
   * Submissions awaiting the caller's decision, oldest first.
   *
   * Derived from the org chart at call time: it lists exactly those submissions the caller could
   * actually act on right now, and is empty for an employee who approves nothing.
   *
   * SHARING: this is the only method that returns other people's records, so it is treated as
   * strictly private. The call FAILS outright if this Gadget is currently shared with anyone — the
   * data is refused rather than shown. If a Gadget needs to be shareable, do not call this;
   * everything else here is the caller's own data and stays shareable.
   *
   * Two limits worth knowing, because neither is what you might assume:
   *
   *  - Calling this does NOT restrict what the Gadget may do afterwards. Deciding on a submission
   *    it returned still works, which is the point of it.
   *  - The check happens per call, against who has access AT THAT MOMENT. Reading the queue on an
   *    unshared Gadget and sharing it afterwards is NOT prevented: later calls will start failing,
   *    but anything this Gadget already rendered from the earlier read stays visible to the new
   *    collaborator. If you display other people's submissions, do not leave them on screen in a
   *    Gadget that may later be shared.
   */
  listPendingApprovals(): Promise<SubmissionRow[]>;

  /**
   * Approve, reject, or return one submission from `listPendingApprovals()`.
   *
   * REQUIRES HUMAN CONFIRMATION, and does not take effect when it returns. This is a manager's
   * sign-off on another employee's pay, so the decision is submitted to the approval queue and
   * only performed once the account holder confirms it — which may be minutes or days later, or
   * never. Returning without throwing means "submitted", not "done".
   *
   * Do NOT poll for the result: the submission is unchanged until the decision is confirmed, and
   * nothing here simulates the pending change, so every read still shows it as `pending`.
   *
   * Re-issuing the SAME decision (same action, same comment) is safe. It is recognised as the one
   * already awaiting confirmation and does not queue a second one, so a retry after a transient
   * failure cannot turn one intent into two sign-offs. Issuing a DIFFERENT decision while one is
   * still awaiting confirmation throws `KINTAI_DECISION_CONFLICT`: the pending one must be
   * confirmed or discarded first, because a human may be looking at it right now.
   *
   * A decision is applied to the submission as it stood when you made it. If the submission is
   * acted on in the meantime — another approver signs it, it is returned, the employee refiles it
   * — the confirmed decision is refused with `KINTAI_STALE_DECISION` instead of being recorded
   * against something its approver never saw. Read the submission again and decide afresh.
   *
   * Authority is checked twice: once now, so an invalid decision is refused immediately rather
   * than put in front of a human, and again when the confirmed decision is applied — the
   * organisation may have changed in between, and a decision that is no longer yours to make is
   * refused at that point instead of being performed.
   *
   * `approve` on the last step approves the submission; earlier steps advance it to the next
   * approver, and `return` sends it back to the employee as a `draft`. Once applied, read the
   * outcome back from `listPendingApprovals()` (an approved or rejected submission leaves your
   * queue) or, for your own submissions, `listMySubmissions()`.
   *
   * Nobody may act on their own submission, and naming a submission the caller is not an approver
   * for throws `KINTAI_NOT_AUTHORIZED` regardless of whether it exists in a different state.
   */
  actOnSubmission(
    submissionId: number,
    action: ApprovalAction,
    comment?: string,
  ): Promise<void>;
}
